# 批量转存「复制到」Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在批量转存页操作列增加「复制到」，把单个文件/文件夹复制到同一账号下多个同级目录；百度、夸克、迅雷都支持，弹框内依次执行并用 toast 反馈。

**Architecture:** 新增 `POST /api/accounts/{id}/files/copy`，body 带源路径与目标目录列表。三网盘适配器各实现 `copy(cookie, path, dest, name)`。服务层校验同级目标、禁止复制到自身/子孙，重名时按 `名称` → `名称(1)` → … 重试。前端新增多选目录弹框，不建后台任务。

**Tech Stack:** Python 3.12、FastAPI、httpx MockTransport、pytest；前端 Vite + React + TypeScript + Tailwind。

**规格：** `docs/superpowers/specs/2026-10-13-copy-to-design.md`

**不要改：** Worker、分享同步、额度逻辑。不要请求真实网盘。不要把远程数据库密码写进仓库。

```python
MAX_TARGETS = 50
MAX_RENAME_ATTEMPTS = 50


def parent_of(path: str) -> str:
    cleaned = path.strip().rstrip("/") or "/"
    if cleaned == "/":
        return "/"
    parent = cleaned.rsplit("/", 1)[0]
    return parent or "/"


def same_parent(paths: list[str]) -> bool:
    if not paths:
        return False
    parents = {parent_of(path) for path in paths}
    return len(parents) == 1


def is_self_or_descendant(source: str, target: str) -> bool:
    src = source.strip().rstrip("/") or "/"
    tgt = target.strip().rstrip("/") or "/"
    if src == "/":
        return False
    return tgt == src or tgt.startswith(src + "/")


def next_copy_name(name: str, *, is_dir: bool, attempt: int) -> str:
    if attempt <= 0:
        return name
    if is_dir:
        return f"{name}({attempt})"
    dot = name.rfind(".")
    if dot <= 0:
        return f"{name}({attempt})"
    return f"{name[:dot]}({attempt}){name[dot:]}"


def validate_targets(source: str, targets: list[str]) -> list[str]:
    cleaned = []
    for raw in targets:
        path = raw.strip().rstrip("/") or "/"
        if path == "/":
            raise ValueError("目标目录无效")
        cleaned.append(path)
    if not cleaned or len(cleaned) > MAX_TARGETS:
        raise ValueError("目标目录数量无效")
    if len(set(cleaned)) != len(cleaned):
        raise ValueError("目标目录重复")
    if not same_parent(cleaned):
        raise ValueError("目标目录必须同级")
    for path in cleaned:
        if is_self_or_descendant(source.strip().rstrip("/") or "/", path):
            raise ValueError("不能复制到自身或子目录")
    return cleaned
```

注意：`is_self_or_descendant("/说明.txt", "/说明.txt/x")` 对文件路径返回 False（文件不是目录前缀语义上的祖先）。实现时：仅当 `source` 在路径上可作为目录前缀时才判子孙——若 source 含扩展名且不以文件夹约定存储，仍用路径前缀规则：`tgt == src or tgt.startswith(src + "/")`。对文件 `/说明.txt`，`/说明.txt/x` 会 `startswith` 为 True。测试里写的是 False——请把测试改为：

```python
assert is_self_or_descendant("/说明.txt", "/说明.txt") is True
assert is_self_or_descendant("/说明.txt", "/其他") is False
```

路径前缀规则统一即可，源是文件时前端不会把源路径当作目标。

def copy(self, cookie: str, path: str, dest: str, new_name: str) -> None:
    wait = time.sleep
    body = {
        "filelist": [_fid(path)],
        "to_pdir_fid": _fid(dest),
        "exclude_fids": [],
    }
    # 若接口支持重命名字段则带上 new_name；否则 copy 后若名称不同再 rename
    created = self._post("/file/copy", cookie, body)
    task_data = created.get("data") if isinstance(created.get("data"), dict) else {}
    task_id = task_data.get("task_id")
    if isinstance(task_id, str) and task_id:
        self._wait_task(cookie, task_id, wait)
    leaf = path.rstrip("/").rsplit("/", 1)[-1]
    # 夸克 path 常为 /{fid}，复制后名称由网盘决定；若 new_name 与源名不同，在目标侧 rename
    # 简化：始终在复制成功后，若需要不同名字则对「目标下刚复制出的项」rename
    # 实现时：若 API 无 rename-on-copy，则 list_dir(dest) 找同名源名再 rename 为 new_name
```

实现约束（写进代码注释即可）：

class CopyIn(BaseModel):
    path: str = Field(min_length=1, max_length=1024)
    fs_id: int | None = None
    name: str = Field(min_length=1, max_length=255)
    is_dir: bool = False
    targets: list[str] = Field(min_length=1, max_length=50)


class CopyTargetResultOut(BaseModel):
    path: str
    name: str
    ok: bool
    error: str | None = None


class CopyOut(BaseModel):
    succeeded: int
    failed: int
    results: list[CopyTargetResultOut]
```

在 `__init__.py` 导出。

路由逻辑：

```python
@router.post("/api/accounts/{account_id}/files/copy", response_model=CopyOut)
def copy_files(...):
    account = owned_drive(...)
    source = require_child_path(body.path)
    name = clean_name(body.name)
    try:
        targets = validate_targets(source, body.targets)
    except ValueError as exc:
        raise HTTPException(status_code=422, detail=str(exc)) from exc
    adapter, cookie = open_adapter(...)
    results = []
    expired = False
    for dest in targets:
        if expired:
            results.append(CopyTargetResultOut(path=dest, name=name, ok=False, error=relogin(account.provider)))
            continue
        placed = None
        error = None
        for attempt in range(MAX_RENAME_ATTEMPTS):
            candidate = next_copy_name(name, is_dir=body.is_dir, attempt=attempt)
            try:
                adapter.copy(cookie, source, dest, candidate)
                placed = candidate
                break
            except DuplicateName:
                continue
            except LoginRejected:
                expired = True
                account.status = "expired"
                ...
                error = relogin(account.provider)
                break
            except PathNotFound:
                error = "目录不存在"
                break
            except TransferFailed:
                error = "复制失败"
                break
        if placed is None and error is None:
            error = "已有同名文件"
        results.append(CopyTargetResultOut(path=dest, name=placed or name, ok=placed is not None, error=error))
    return CopyOut(succeeded=sum(1 for r in results if r.ok), failed=sum(1 for r in results if not r.ok), results=results)
```
